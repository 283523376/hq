# WMS 引用完整性与删除守卫 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为永鑫WMS 全部 7 张 `wms_` 表补齐应用层删除守卫与批量删除事务,并提供前端「预检 + 禁止弹窗」统一交互,闭环 TC-BD002-006。

**Architecture:** 每个 Service 暴露 `checkDelete(id)`（只读收集全部 blocker）与守卫化的 `removeXxx`（不可删即抛 `RuntimeException`，文案逐字对齐测试用例）；5 个 Controller 增加只读预检端点 `checkDelete` / `checkDeleteBatch`；所有 `removeBatch` 下沉到 Service 层并加 `@Transactional` 循环复用单条守卫，杜绝部分成功。前端新建 `useWmsDeleteCheck` composable，把 5 个页面的删除入口从 `popConfirm` 气泡改为「预检 → 禁止弹窗 / 确认弹窗」。数据库零 DDL。

**Tech Stack:** Spring Boot 4.1 / MyBatis-Plus 3.5（`@TableLogic` deleted_at 时间戳）/ JUnit 5 + Mockito（spy + `WmsTestSupport.inject` 反射注入，不启 Spring 不连库）/ Vue 3 + vben（`useMessage` 的 `createConfirm` / `createErrorModal`）。

**Spec:** `docs/superpowers/specs/2026-10-16-yx-wms-引用完整性与删除守卫-design.md`（本计划与其同步提交；分类不设批量预检的收窄已回写 spec §3.2）

## Review Focus

1. **预检与删除之间的并发窗口**——预检通过后、删除前他人新建了订单。期望：后端守卫兜底拒绝，绝不能因前端预检通过而跳过后端检查。→ Task 1/3 的守卫单测 + Task 14 端到端步骤 6。
2. **批量删除部分成功**——中途遇守卫必须整体回滚。单测只能验证「循环复用 + 异常传播」（无 Spring 事务上下文），`@Transactional` 的实际回滚由 Task 14 端到端步骤 3（API 直调）验证，不许略过。
3. **文案被污染**——`JeecgBootExceptionHandler` 会给未捕获异常加前缀，导致 TC-BD002-006「逐字命中」验收失败。→ Task 1/2/3 的 controller try/catch + Task 14 端到端步骤 1。
4. **不存在 / 已逻辑删除的 id**——`checkDelete` 对查不到的 id 必须返回「可删」（与现状 `removeById` 对不存在 id 无操作一致），不得 NPE、不得误报。→ Task 1 步骤 1 的 `checkDelete_shouldBeDeletableForUnknownId`。
5. **守卫计数包含已删引用行**——被逻辑删除的订单不该阻止删供应商。依赖 `@TableLogic` 自动条件；wrapper 层断言不到（条件由 MP 在 mapper 层拼接），靠「不手写 deleted_at 条件」这条全局约束 + 端到端兜底。

```java
    @Autowired
    private WmsOrderMapper wmsOrderMapper;

    @Override
    public WmsDeleteCheckVO checkDelete(String id) {
        WmsDeleteCheckVO vo = new WmsDeleteCheckVO();
        if (oConvertUtils.isEmpty(id)) {
            return vo;
        }
        long orderCount = wmsOrderMapper.selectCount(
                new QueryWrapper<WmsOrder>().eq("supplier_id", id));
        if (orderCount > 0) {
            // 样本取最近 5 单的订单号，供前端禁止弹窗展示「被哪些单占用」
            List<WmsOrder> recent = wmsOrderMapper.selectList(new QueryWrapper<WmsOrder>()
                    .eq("supplier_id", id).orderByDesc("create_time").last("limit 5"));
            vo.addBlocker("存在关联订单", orderCount,
                    recent.stream().map(WmsOrder::getOrderNo).collect(Collectors.toList()));
        }
        return vo;
    }

    @Override
    public WmsBatchDeleteCheckVO checkDeleteBatch(List<String> ids) {
        WmsBatchDeleteCheckVO vo = new WmsBatchDeleteCheckVO();
        if (ids == null || ids.isEmpty()) {
            return vo;
        }
        for (String id : ids) {
            WmsDeleteCheckVO check = checkDelete(id);
            if (!check.isDeletable()) {
                WmsSupplier supplier = this.getById(id);
                vo.addBlocked(id,
                        supplier == null ? id : supplier.getSupplierCode(),
                        check.getBlockers().stream()
                                .map(WmsDeleteCheckVO.Blocker::getMessage)
                                .collect(Collectors.joining("；")));
            }
        }
        return vo;
    }

    @Override
    public void removeSupplier(String id) {
        WmsDeleteCheckVO check = checkDelete(id);
        if (!check.isDeletable()) {
            // 文案逐字对齐 TC-BD002-006 预期结果，前端与测试都按原文匹配
            throw new RuntimeException("该供应商存在关联订单，不允许删除");
        }
        this.removeById(id);
    }

    /**
     * 批量删除必须整体在一个事务里：循环复用单条守卫，第 N 条被拒时前 N-1 条一并回滚——
     * 否则「部分成功」会让接口报错但数据已删一半（错误的状态结论，与设备 bind 同一个教训）。
     * 自调用说明：事务注解在本方法上、由 controller 经代理进入，内部 this.removeSupplier 不经过代理无碍。
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public void removeBatch(List<String> ids) {
        if (ids != null && !ids.isEmpty()) {
            for (String id : ids) {
                removeSupplier(id);
            }
        }
    }
```

```java
        } catch (Exception e) {
            log.error(e.getMessage(), e);
            return Result.error(e.getMessage());
```

```java
    public WmsDeleteCheckVO checkDelete(String id) {
        WmsDeleteCheckVO vo = new WmsDeleteCheckVO();
        if (oConvertUtils.isEmpty(id)) {
            return vo;
        }
        long locationCount = wmsLocationMapper.selectCount(
                new QueryWrapper<WmsLocation>().eq("warehouse_id", id));
        if (locationCount > 0) {
            vo.addBlocker("该库区下存在库位", locationCount, Collections.emptyList());
        }
        long deviceCount = wmsDeviceMapper.selectCount(
                new QueryWrapper<WmsDevice>().eq("warehouse_id", id));
        if (deviceCount > 0) {
            vo.addBlocker("该库区已绑定设备", deviceCount, Collections.emptyList());
        }
        return vo;
    }

    @Override
    public WmsBatchDeleteCheckVO checkDeleteBatch(List<String> ids) {
        WmsBatchDeleteCheckVO vo = new WmsBatchDeleteCheckVO();
        if (ids == null || ids.isEmpty()) {
            return vo;
        }
        for (String id : ids) {
            WmsDeleteCheckVO check = checkDelete(id);
            if (!check.isDeletable()) {
                WmsWarehouse warehouse = this.getById(id);
                vo.addBlocked(id,
                        warehouse == null ? id : warehouse.getWarehouseCode(),
                        check.getBlockers().stream()
                                .map(WmsDeleteCheckVO.Blocker::getMessage)
                                .collect(Collectors.joining("；")));
            }
        }
        return vo;
    }

    @Override
    public void removeWarehouse(String id) {
        WmsDeleteCheckVO check = checkDelete(id);
        if (!check.isDeletable()) {
            // blocker 顺序即优先级：库位先于设备，与既有文案一致
            throw new RuntimeException(check.getBlockers().get(0).getMessage() + "，不允许删除");
        }
        this.removeById(id);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void removeBatch(List<String> ids) {
        if (ids != null && !ids.isEmpty()) {
```

`useWmsDeleteCheck.ts` 全文：

```typescript
import { h } from 'vue';
import { createConfirm, createErrorModal } from '/@/hooks/web/useMessage';

/** 对齐后端 org.jeecg.modules.wms.vo.WmsDeleteCheckVO */
export interface WmsDeleteBlocker {
  message: string;
  count: number;
  samples: string[];
}

export interface WmsDeleteCheck {
  deletable: boolean;
  blockers: WmsDeleteBlocker[];
}

/** 对齐后端 org.jeecg.modules.wms.vo.WmsBatchDeleteCheckVO */
export interface WmsBatchDeleteCheck {
  deletable: boolean;
  blockedItems: { id: string; code: string; reason: string }[];
}

export interface WmsDeleteCheckApis {
  checkDelete: (params: { id: string }) => Promise<WmsDeleteCheck>;
  checkDeleteBatch?: (params: { ids: string }) => Promise<WmsBatchDeleteCheck>;
  remove: (params: { id: string }, handleSuccess: () => void) => Promise<any>;
  removeBatch: (params: { ids: string }, handleSuccess: () => void) => Promise<any>;
}

/** 弹「无法删除」禁止框：blockers 逐条一行（树组件与列表 composable 共用）。 */
export function showDeleteBlocked(blockers: WmsDeleteBlocker[] | undefined, title = '无法删除') {
  const lines = (blockers ?? []).map((b) => {
    const sample = b.samples?.length ? `：${b.samples.join('、')}` : '';
    return `${b.message}（${b.count}）${sample}`;
  });
  createErrorModal({
    title,
    content: lines.length
      ? h('div', null, lines.map((l) => h('div', l)))
      : '该记录存在关联数据，不允许删除',
  });
}

/**
 * WMS 删除守卫统一交互（spec 2026-10-16 §4）：
 * 点删除先预检——可删才弹确认框，不可删弹禁止框列出原因；
 * 后端守卫仍在删除时兜底（预检与删除之间存在并发窗口）。
 */
export function useWmsDeleteCheck(apis: WmsDeleteCheckApis, onSuccess: () => void) {
  async function confirmDelete(record: { id: string }) {
    const check: WmsDeleteCheck = await apis.checkDelete({ id: record.id });
    if (!check?.deletable) {
      showDeleteBlocked(check?.blockers);
      return;
    }
    createConfirm({
      iconType: 'warning',
      title: '确定删除吗?',
      content: '删除后数据保留删除时间戳，可在数据库中恢复',
      onOk: async () => {
        await apis.remove({ id: record.id }, onSuccess);
      },
    });
  }

  async function confirmBatchDelete(ids: string[]) {
    if (!ids.length) {
      return;
    }
    if (apis.checkDeleteBatch) {
      const check = await apis.checkDeleteBatch({ ids: ids.join(',') });
      if (!check?.deletable) {
        const lines = (check?.blockedItems ?? []).map((it) => `${it.code}：${it.reason}`);
        createErrorModal({
          title: '以下记录无法删除',
          content: lines.length
            ? h('div', null, lines.map((l) => h('div', l)))
            : '所选记录存在关联数据，不允许删除',
        });
        return;
      }
    }
    createConfirm({
      iconType: 'warning',
      title: `确定删除选中的 ${ids.length} 条记录吗?`,
      content: '删除后数据保留删除时间戳，可在数据库中恢复',
      onOk: async () => {
        await apis.removeBatch({ ids: ids.join(',') }, onSuccess);
      },
    });
  }

  return { confirmDelete, confirmBatchDelete };
}
```
