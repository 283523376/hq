# YX-WMS-BD-004 权限与系统配置 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让系统管理员能按分组维护 WMS 系统参数（库存预警阈值 / 上位系统同步范围 / 不良品存放规则 / 设备告警阈值），支持校验冲突提示与回滚到上一版本，并让设备两项阈值真正生效。

**Architecture:** 两张表——`wms_system_config`（值，键值对）+ `wms_system_config_history`（每次变更追加一行，供留痕与回滚）。配置项的元数据（中文名/分组/类型/校验规则/默认值）写进 Java 枚举 `WmsConfigKey`，校验器 `WmsConfigValidator` 是**纯函数**（可穷举单测）。保存 = 单事务内「全量校验 → upsert 值 → 追加历史」；回滚 = 取该组每个键的次新版本写回并再次追加历史（不删历史）。冲突时 `Result.message` 逐字等于 `配置项存在冲突，请检查`，明细放在同一次响应的 `Result.result.conflicts[]`。

**Tech Stack:** Spring Boot 4.1 / Java 17 / MyBatis-Plus（`ServiceImpl` + `QueryWrapper`，`@TableLogic` 用 `deleted_at`）/ JUnit 5 + Mockito（`spy` + `WmsTestSupport.inject`，不启 Spring 不连库）/ Vue 3 + Ant Design Vue（**不用 BasicForm**，见 Task 9 说明）/ Flyway（`V3.9.5_10`）。

## Review Focus

1. **同一分组被并发保存 → 历史版本号可能重复。** 版本号取「该键当前最大值 + 1」，而 READ COMMITTED 下这**不是串行**的：两个并发保存会算出同一个版本号。期望行为：不丢数据、且回滚的结果**确定**（同样输入永远回滚到同一版本）。→ Task 6 的 `rollback` 排序必须带 tie-break（`version_no DESC, create_time DESC, id DESC`），并由 `history_shouldOrderDeterministicallyOnDuplicateVersion` 固化。
2. **未保存就切换分组 → 改动静默丢失。** 期望：切走前有提示（不是静默丢失）。→ Task 9 前端 `handleGroupChange` + Task 11 端到端步骤 6。
3. **回滚到「当前校验规则下已不合法」的旧值。** 期望：拒绝回滚并说明原因，**不得**把非法值写进主表。→ Task 6 的 `rollback_shouldRejectWhenPreviousValueInvalid`。
4. **`getInt` 读到脏值（空串/非数字/≤0/行不存在）。** 期望：返回兜底值 + WARN，**绝不抛异常**（配置坏了不能连离线扫描一起弄死）。→ Task 8 的四个 `getInt` 用例。
5. **冲突响应里明细为空。** 期望：只要 `success=false` 且文案是冲突文案，`conflicts` 必须非空（否则用户只看到一句无从下手的话）。→ Task 5 的 `save_shouldCarryConflictsPayload`。

```java
package org.jeecg.modules.wms.constant;

import java.util.Arrays;
import java.util.List;

/**
 * 系统配置分组（蓝图 YX-WMS-BD-004 的步骤 3~6）。
 *
 * <p>顺序即前端左侧菜单顺序，故显式给出 {@link #ordered()}，不依赖声明顺序。
 *
 * @author 永鑫WMS
 */
public enum WmsConfigGroup {

    INVENTORY("inventory", "库存预警阈值"),
    SYNC("sync", "上位系统同步范围"),
    DEFECT("defect", "不良品存放规则"),
    DEVICE("device", "设备告警阈值");

    private final String code;
    private final String name;

    WmsConfigGroup(String code, String name) {
        this.code = code;
        this.name = name;
    }

    public String getCode() {
        return code;
    }

    public String getName() {
        return name;
    }

    /** 按 code 取分组；未知 code 直接拒绝，避免静默落到某个默认分组。 */
    public static WmsConfigGroup of(String code) {
        for (WmsConfigGroup group : values()) {
            if (group.code.equals(code)) {
                return group;
            }
        }
        throw new IllegalArgumentException("未知配置分组：" + code);
    }

    /** 前端菜单顺序。 */
    public static List<WmsConfigGroup> orderedAsList() {
        return Arrays.asList(INVENTORY, SYNC, DEFECT, DEVICE);
    }

    public static WmsConfigGroup[] ordered() {
        return orderedAsList().toArray(new WmsConfigGroup[0]);
    }
}
```

```java
package org.jeecg.modules.wms.constant;

import org.jeecg.modules.wms.vo.WmsConfigConflict;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

/**
 * 配置校验器（**纯函数**，不依赖 Spring、不碰数据库，故可穷举测试）。
 *
 * <p>校验三件事，缺一不可：
 * <ol>
 *   <li><b>归属</b>：提交的键必须属于所声明的分组（否则前端能把别的组的值写进来）；</li>
 *   <li><b>类型与值域</b>：按 {@link WmsConfigKey#getType()} 解析，再按项判值域；</li>
 *   <li><b>组内自洽</b>：库存三项必须满足 最低 ≤ 安全 ≤ 最高。</li>
 * </ol>
 *
 * <p>只「报」不「抛」业务冲突（返回列表）；只有调用方传了未知分组这种**编程错误**才抛
 * {@link IllegalArgumentException} ——把它伪装成业务冲突会让真 bug 混在用户输入里。
 *
 * @author 永鑫WMS
 */
public final class WmsConfigValidator {

    /** 不良品默认仓的编码长度上限，与库区编码同宽（location_code 列宽 64）。 */
    public static final int CODE_MAX_LENGTH = 64;

    private WmsConfigValidator() {
    }

    /**
     * 校验某分组的全部配置值。
     *
     * @param groupCode 分组 code
     * @param values    该分组的全部键值（**全组提交**，缺项即冲突）
     * @return 冲突列表；空列表表示通过
     */
    public static List<WmsConfigConflict> validate(String groupCode, Map<String, String> values) {
        WmsConfigGroup group = WmsConfigGroup.of(groupCode);
        Map<String, String> safeValues = values == null ? Map.of() : values;
        List<WmsConfigConflict> conflicts = new ArrayList<>();

        // 1) 越组写入：提交里出现了不属于本组的键
        Set<String> groupKeys = new HashSet<>();
        for (WmsConfigKey item : WmsConfigKey.ofGroup(group)) {
            groupKeys.add(item.getKey());
        }
        for (String submitted : safeValues.keySet()) {
            if (!groupKeys.contains(submitted)) {
                WmsConfigKey foreign = WmsConfigKey.of(submitted);
                conflicts.add(new WmsConfigConflict(submitted, foreign.getLabel(),
                        "配置项不属于分组「" + group.getName() + "」"));
            }
        }

        // 2) 逐项：缺项 / 类型 / 值域
        for (WmsConfigKey item : WmsConfigKey.ofGroup(group)) {
            if (!safeValues.containsKey(item.getKey())) {
                conflicts.add(new WmsConfigConflict(item.getKey(), item.getLabel(), "配置项缺失，保存需提交整组"));
                continue;
            }
            String raw = safeValues.get(item.getKey());
            String reason = checkValue(item, raw);
            if (reason != null) {
                conflicts.add(new WmsConfigConflict(item.getKey(), item.getLabel(), reason));
            }
        }

        // 3) 组内自洽（只在三项都能解析为整数时才判，避免与上面的类型错误重复报）
        if (group == WmsConfigGroup.INVENTORY && conflicts.isEmpty()) {
            int min = parseInt(safeValues.get(WmsConfigKey.INV_STOCK_MIN.getKey()));
            int safe = parseInt(safeValues.get(WmsConfigKey.INV_STOCK_SAFE.getKey()));
            int max = parseInt(safeValues.get(WmsConfigKey.INV_STOCK_MAX.getKey()));
            if (min > safe) {
                conflicts.add(new WmsConfigConflict(WmsConfigKey.INV_STOCK_MIN.getKey(),
                        WmsConfigKey.INV_STOCK_MIN.getLabel(),
                        "最低库存(" + min + ") 不能大于 安全库存(" + safe + ")"));
            } else if (safe > max) {
                conflicts.add(new WmsConfigConflict(WmsConfigKey.INV_STOCK_SAFE.getKey(),
                        WmsConfigKey.INV_STOCK_SAFE.getLabel(),
                        "安全库存(" + safe + ") 不能大于 最高库存(" + max + ")"));
            }
        }
        return conflicts;
    }

    /** 单值检查；返回 null 表示通过。 */
    private static String checkValue(WmsConfigKey item, String raw) {
        String value = raw == null ? "" : raw.trim();
        switch (item.getType()) {
            case INT: {
                if (!value.matches("-?\\d+")) {
                    return "必须为整数，实际：" + (raw == null ? "(空)" : raw);
                }
                long parsed = Long.parseLong(value);
                if (item.getGroup() == WmsConfigGroup.INVENTORY && parsed < 0) {
                    return "不能为负数，实际：" + value;
                }
                if (item.getGroup() == WmsConfigGroup.DEVICE && parsed <= 0) {
                    return "必须大于 0，实际：" + value;
                }
                return null;
            }
            case BOOL:
                return "true".equals(value) || "false".equals(value) ? null : "必须为 true 或 false，实际：" + value;
            case MULTI: {
                if (value.isEmpty()) {
                    return "不能为空";
                }
                List<String> allowed = item.getOptions();
                for (String part : value.split(",")) {
                    String option = part.trim();
                    if (option.isEmpty()) {
                        continue;
                    }
                    if (allowed == null || !allowed.contains(option)) {
                        return "取值不在候选范围（" + String.join("/", allowed == null ? List.of() : allowed) + "），实际：" + option;
                    }
                }
                return null;
            }
            case TEXT: {
                if (item == WmsConfigKey.DEFECT_WAREHOUSE_CODE && value.length() > CODE_MAX_LENGTH) {
                    return "长度不能超过 " + CODE_MAX_LENGTH + " 字符，实际：" + value.length();
                }
                return null;
            }
            default:
                return null;
        }
    }

    private static int parseInt(String raw) {
        try {
            return Integer.parseInt(raw.trim());
        } catch (Exception e) {
            return 0;
        }
    }
}
```

```java
import org.jeecg.modules.wms.entity.WmsSystemConfig;
```

```java
import org.jeecg.modules.wms.vo.WmsConfigGroupVO;
import org.jeecg.modules.wms.vo.WmsConfigItemVO;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * @Description: 系统配置
 * @Author: 永鑫WMS
 * @Date: 2026-10-16
 */
@Slf4j
@Service
public class WmsSystemConfigServiceImpl extends ServiceImpl<WmsSystemConfigMapper, WmsSystemConfig>
        implements IWmsSystemConfigService {

    @Autowired
    private WmsSystemConfigHistoryMapper wmsSystemConfigHistoryMapper;

    @Override
    public List<WmsConfigGroupVO> definitions() {
        Map<String, WmsSystemConfig> stored = loadStoredMap();
        List<WmsConfigGroupVO> groups = new ArrayList<>();
        for (WmsConfigGroup group : WmsConfigGroup.orderedAsList()) {
            WmsConfigGroupVO vo = new WmsConfigGroupVO();
            vo.setCode(group.getCode());
            vo.setName(group.getName());
            List<WmsConfigItemVO> items = new ArrayList<>();
            for (WmsConfigKey key : WmsConfigKey.ofGroup(group)) {
                items.add(toItemVO(key, stored.get(key.getKey())));
            }
            vo.setItems(items);
            groups.add(vo);
        }
        return groups;
    }

    @Override
    public List<WmsConfigItemVO> listByGroup(String groupCode) {
        WmsConfigGroup group = WmsConfigGroup.of(groupCode);
        Map<String, WmsSystemConfig> stored = loadStoredMap();
        List<WmsConfigItemVO> items = new ArrayList<>();
        for (WmsConfigKey key : WmsConfigKey.ofGroup(group)) {
            items.add(toItemVO(key, stored.get(key.getKey())));
        }
        return items;
    }

    @Override
    public int currentVersionNo(String key) {
        WmsSystemConfigHistory latest = wmsSystemConfigHistoryMapper.selectOne(
                new QueryWrapper<WmsSystemConfigHistory>()
                        .eq("config_key", key)
                        .orderByDesc("version_no", "create_time", "id")
                        .last("limit 1"));
        return latest == null || latest.getVersionNo() == null ? 0 : latest.getVersionNo();
    }

    /** 一次性把在存配置读成 map，避免逐项查库。 */
    private Map<String, WmsSystemConfig> loadStoredMap() {
        Map<String, WmsSystemConfig> map = new HashMap<>();
        for (WmsSystemConfig row : this.list(new QueryWrapper<>())) {
            map.put(row.getConfigKey(), row);
        }
        return map;
    }

    private WmsConfigItemVO toItemVO(WmsConfigKey key, WmsSystemConfig stored) {
        WmsConfigItemVO vo = new WmsConfigItemVO();
        vo.setKey(key.getKey());
        vo.setLabel(key.getLabel());
        vo.setType(key.getType().name());
        // 库里没有该键时回落元数据默认值：新环境首次打开页面不能是空白
        vo.setValue(stored == null || stored.getConfigValue() == null
                ? key.getDefaultValue() : stored.getConfigValue());
        vo.setOptions(key.getOptions());
        int version = currentVersionNo(key.getKey());
        vo.setVersionNo(version);
        vo.setCanRollback(version >= 2);
        return vo;
    }
}
```

```java
    @Transactional(rollbackFor = Exception.class)
    public void save(WmsConfigSaveDTO dto) {
        if (dto == null || dto.getGroup() == null) {
            throw new IllegalArgumentException("缺少配置分组");
        }
        Map<String, String> values = new LinkedHashMap<>();
        if (dto.getItems() != null) {
            for (WmsConfigItemDTO item : dto.getItems()) {
                if (item != null && item.getKey() != null) {
                    values.put(item.getKey(), item.getValue());
                }
            }
        }

        // 1) 先校验，全组通过才动库。校验失败时下面两个写方法一次都不会被调用。
        List<WmsConfigConflict> conflicts = WmsConfigValidator.validate(dto.getGroup(), values);
        if (!conflicts.isEmpty()) {
            throw new WmsConfigConflictException(conflicts);
        }

        // 2) upsert 主表 + 3) 追加历史（同一事务：写历史失败则 upsert 一起回滚，不会出现「值改了没留痕」）
        Map<String, WmsSystemConfig> stored = loadStoredMap();
        int next = 0;
        for (Map.Entry<String, String> entry : values.entrySet()) {
            WmsConfigKey meta = WmsConfigKey.of(entry.getKey());
            String value = entry.getValue() == null ? "" : entry.getValue().trim();

            WmsSystemConfig existing = stored.get(entry.getKey());
            if (existing == null) {
                WmsSystemConfig row = new WmsSystemConfig()
                        .setConfigKey(meta.getKey())
                        .setConfigValue(value)
                        .setConfigDesc(meta.getLabel());
                WmsUtils.fillCreate(row);
                this.save(row);
            } else {
                existing.setConfigValue(value);
                WmsUtils.fillUpdate(existing);
                this.updateById(existing);
            }

            WmsSystemConfigHistory history = new WmsSystemConfigHistory();
            history.setConfigKey(meta.getKey());
            history.setConfigValue(value);
            history.setVersionNo(currentVersionNo(meta.getKey()) + 1);
            history.setChangeType(WmsConstants.CONFIG_CHANGE_SAVE);
            WmsUtils.fillCreate(history);
            wmsSystemConfigHistoryMapper.insert(history);
            next++;
        }
        log.info("系统配置已保存：分组={}，项数={}", dto.getGroup(), next);
```

```java
    @Override
    @Transactional(rollbackFor = Exception.class)
    public void rollback(WmsConfigRollbackDTO dto) {
        if (dto == null || dto.getGroup() == null) {
            throw new IllegalArgumentException("缺少配置分组");
        }
        WmsConfigGroup group = WmsConfigGroup.of(dto.getGroup());

        // 1) 逐键取「次新」版本（当前值是最大版本，上一版本是第 2 条）
        Map<String, String> previous = new LinkedHashMap<>();
        for (WmsConfigKey key : WmsConfigKey.ofGroup(group)) {
            List<WmsSystemConfigHistory> rows = recentHistory(key.getKey(), 2);
            if (rows.size() >= 2) {
                previous.put(key.getKey(), rows.get(1).getConfigValue());
            }
        }
        if (previous.isEmpty()) {
            throw new RuntimeException("当前分组没有可回滚的历史版本");
        }

        // 2) 回滚目标值必须仍能过校验：规则中途加严时，旧值可能已非法 ——
        //    宁可拒绝，也不要把非法值写进主表再让人去收拾
        List<WmsConfigConflict> conflicts = WmsConfigValidator.validate(group.getCode(), previous);
        if (!conflicts.isEmpty()) {
            throw new WmsConfigConflictException(conflicts);
        }

        // 3) 写回主表 + 追加 ROLLBACK 历史（不删旧历史，审计链完整）
        Map<String, WmsSystemConfig> stored = loadStoredMap();
        for (Map.Entry<String, String> entry : previous.entrySet()) {
            WmsConfigKey meta = WmsConfigKey.of(entry.getKey());
            String value = entry.getValue() == null ? "" : entry.getValue();

            WmsSystemConfig existing = stored.get(entry.getKey());
            if (existing == null) {
                WmsSystemConfig row = new WmsSystemConfig()
                        .setConfigKey(meta.getKey()).setConfigValue(value).setConfigDesc(meta.getLabel());
                WmsUtils.fillCreate(row);
                this.save(row);
            } else {
                existing.setConfigValue(value);
                WmsUtils.fillUpdate(existing);
                this.updateById(existing);
            }

            WmsSystemConfigHistory history = new WmsSystemConfigHistory();
            history.setConfigKey(meta.getKey());
            history.setConfigValue(value);
            history.setVersionNo(currentVersionNo(meta.getKey()) + 1);
            history.setChangeType(WmsConstants.CONFIG_CHANGE_ROLLBACK);
            WmsUtils.fillCreate(history);
            wmsSystemConfigHistoryMapper.insert(history);
        }
        log.info("系统配置已回滚：分组={}，项数={}", dto.getGroup(), previous.size());
    }

    @Override
    public List<WmsSystemConfigHistory> history(String key) {
        return wmsSystemConfigHistoryMapper.selectList(new QueryWrapper<WmsSystemConfigHistory>()
                .eq("config_key", key)
                .orderByDesc("version_no", "create_time", "id"));
    }

    /**
     * 取某键最近 n 条历史（版本降序）。
     *
     * <p>**排序必须带 tie-break**（version_no → create_time → id）：版本号取「最大值 + 1」，
     * 而 READ COMMITTED 下并发保存可能算出同一个版本号，只按 version_no 排序会让回滚结果不确定。
     */
    private List<WmsSystemConfigHistory> recentHistory(String key, int limit) {
        return wmsSystemConfigHistoryMapper.selectList(new QueryWrapper<WmsSystemConfigHistory>()
                .eq("config_key", key)
                .orderByDesc("version_no", "create_time", "id")
                .last("limit " + limit));
    }
```

```java
    @Override
    public int getInt(String key, int fallback) {
        try {
            WmsConfigKey meta = WmsConfigKey.of(key);
            WmsSystemConfig stored = this.getOne(new QueryWrapper<WmsSystemConfig>()
                    .eq("config_key", meta.getKey()), false);
            if (stored == null || oConvertUtils.isEmpty(stored.getConfigValue())) {
                return fallback;
            }
            int parsed = Integer.parseInt(stored.getConfigValue().trim());
            if (parsed <= 0) {
                log.warn("系统配置[{}]的值 {} 非正数，改用兜底值 {}", key, parsed, fallback);
                return fallback;
            }
            return parsed;
        } catch (IllegalArgumentException e) {
            // 未知键属于编程错误，但读取路径不能因此崩：记 WARN 用兜底
            log.warn("读取了未知配置键[{}]，改用兜底值 {}", key, fallback);
            return fallback;
        } catch (Exception e) {
            // 配置损坏（非数字等）不得让调用方（离线扫描任务）失败 —— 监控链路断了比阈值不准严重得多
            log.warn("系统配置[{}]的值不合法（{}），改用兜底值 {}", key, e.getMessage(), fallback);
            return fallback;
        }
    }
```

**为什么不用 `BasicForm`**：本页的表单结构是**运行时**由 `/definitions` 决定的（分组数、项数、控件类型都不固定），
而 `BasicForm` 的 `schemas` 在 `useForm` 时就要确定；且本项目已记录过它的两个坑
（`getFieldsValue` 非响应式、注册前返回 `undefined` 会让渲染期抛错导致整片空白）。
本页改用原生 `<a-form>` + `v-for`，完全响应式，也便于给每组的「保存」按钮做独立 loading。

```typescript
import { defHttp } from '/@/utils/http/axios';

enum Api {
  definitions = '/wms/systemConfig/definitions',
  list = '/wms/systemConfig/list',
  save = '/wms/systemConfig/save',
  rollback = '/wms/systemConfig/rollback',
  history = '/wms/systemConfig/history',
}

/** 分组结构 + 全部组的值（页面首载唯一入口） */
export const getDefinitions = () => defHttp.get({ url: Api.definitions });

/** 单组刷新 */
export const listGroupConfig = (params: { group: string }) => defHttp.get({ url: Api.list, params });

/**
 * 保存某组。返回原始报文以便读取冲突明细：
 * 冲突时 success=false、message 为固定文案、result.conflicts 为原因列表。
 */
export const saveGroupConfig = (params) =>
  defHttp.post({ url: Api.save, params }, { isTransformResponse: false });

export const rollbackGroupConfig = (params: { group: string }) =>
  defHttp.post({ url: Api.rollback, params }, { isTransformResponse: false });

/** 版本流水（留痕可查证） */
export const listConfigHistory = (params: { key: string }) => defHttp.get({ url: Api.history, params });
```

```vue
<script lang="ts" name="wms-system-config" setup>
  import { computed, onMounted, reactive, ref } from 'vue';
  import { Modal } from 'ant-design-vue';
  import { useMessage } from '/@/hooks/web/useMessage';
  import {
    getDefinitions,
    saveGroupConfig,
    rollbackGroupConfig,
  } from './systemConfig.api';

  const { createMessage } = useMessage();

  const loading = ref(false);
  const saving = ref(false);
  const rolling = ref(false);
  /** 分组结构 + 值，全部来自 /definitions */
  const groups = ref<any[]>([]);
  const currentGroup = ref('');
  /** 表单当前值：key -> string（与后端一致，全部按字符串传） */
  const formValues = reactive<Record<string, string>>({});
  /** 上次加载/保存后的快照，用于判断「有未保存的修改」 */
  let snapshot: Record<string, string> = {};

  const currentItems = computed(() => groups.value.find((g) => g.code === currentGroup.value)?.items ?? []);
  const currentGroupName = computed(() => groups.value.find((g) => g.code === currentGroup.value)?.name ?? '');
  const dirty = computed(() => JSON.stringify(formValues) !== JSON.stringify(snapshot));
  const canRollbackCurrent = computed(() => currentItems.value.some((i: any) => i.canRollback));

  function multiValue(item: any): string[] {
    const raw = formValues[item.key];
    return raw ? raw.split(',').filter((s: string) => s) : [];
  }

  function intValue(item: any): number | undefined {
    const raw = formValues[item.key];
    return raw === '' || raw === undefined ? undefined : Number(raw);
  }

  /** 把某组的项灌进表单并重置快照 */
  function loadGroupIntoForm(items: any[]) {
    for (const key of Object.keys(formValues)) {
      delete formValues[key];
    }
    for (const item of items) {
      formValues[item.key] = item.value ?? '';
    }
    snapshot = JSON.parse(JSON.stringify(formValues));
  }

  async function loadAll() {
    loading.value = true;
    try {
      const res: any = await getDefinitions();
      const list = res?.result ?? res ?? [];
      groups.value = list;
      if (!currentGroup.value && list.length) {
        currentGroup.value = list[0].code;
      }
      const current = list.find((g: any) => g.code === currentGroup.value) ?? list[0];
      if (current) {
        loadGroupIntoForm(current.items);
      }
    } finally {
      loading.value = false;
    }
  }

  onMounted(loadAll);

  /** 切换分组前提示未保存的改动（Review Focus 2：不能静默丢失） */
  function handleGroupChange(e: any) {
    const next = e?.key;
    if (!next || next === currentGroup.value) {
      return;
    }
    if (dirty.value) {
      Modal.confirm({
        title: '有未保存的修改',
        content: '切换分组会丢失当前修改，确定继续吗？',
        okText: '继续切换',
        cancelText: '留在本页',
        onOk: () => {
          currentGroup.value = next;
          loadGroupIntoForm(currentItems.value);
        },
      });
      return;
    }
    currentGroup.value = next;
    loadGroupIntoForm(currentItems.value);
  }

  function buildPayload() {
    return {
      group: currentGroup.value,
      items: currentItems.value.map((i: any) => ({ key: i.key, value: formValues[i.key] ?? '' })),
    };
  }

  /** 冲突时：固定文案 + 明细列表（明细来自 result.conflicts） */
  function showConflicts(message: string, conflicts: any[]) {
    Modal.error({
      title: message,
      content: conflicts?.length
        ? conflicts.map((c: any) => `${c.name || c.key}：${c.reason}`).join('；')
        : '请检查各项取值的取值范围与相互关系',
    });
  }

  async function handleSave() {
    saving.value = true;
    try {
      const res: any = await saveGroupConfig(buildPayload());
      // isTransformResponse:false → 拿到原始报文，自行判断
      if (res?.success === false) {
        showConflicts(res.message, res.result?.conflicts ?? []);
        return;
      }
      createMessage.success('保存成功');
      await loadAll();
    } finally {
      saving.value = false;
    }
  }

  function handleRollback() {
    Modal.confirm({
      title: '回滚确认',
      content: `将「${currentGroupName.value}」整组回退到上一版本，当前值会被覆盖。确定继续吗？`,
      okText: '确认回滚',
      cancelText: '取消',
      onOk: async () => {
        rolling.value = true;
        try {
          const res: any = await rollbackGroupConfig({ group: currentGroup.value });
          if (res?.success === false) {
            showConflicts(res.message, res.result?.conflicts ?? []);
            return;
          }
          createMessage.success('回滚成功');
          await loadAll();
        } finally {
```
